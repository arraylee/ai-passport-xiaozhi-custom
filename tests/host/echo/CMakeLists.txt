set(CMAKE_C_STANDARD 11)

set(XZ "${REPO}/components/platform/xiaozhi")
set(SPX "${XZ}/vendor/speexdsp")
add_library(echo_kernel STATIC "${SPX}/libspeexdsp/mdf.c" "${SPX}/libspeexdsp/fftwrap.c" "${SPX}/libspeexdsp/kiss_fft.c" "${SPX}/libspeexdsp/kiss_fftr.c")
target_include_directories(echo_kernel PRIVATE "${SPX}/include" "${SPX}/libspeexdsp")
target_compile_definitions(echo_kernel PRIVATE FIXED_POINT USE_KISS_FFT EXPORT= OS_SUPPORT_CUSTOM)
# The pinned integer DSP intentionally uses signed two's-complement shifts.
# Check all other undefined behavior; our wrapper and tests keep full UBSan.
target_compile_options(echo_kernel PRIVATE -O2 -fwrapv -fsanitize=undefined -fno-sanitize=shift -fno-sanitize-recover=all)
add_library(echo_owner STATIC "${XZ}/xz_echo.c")
target_include_directories(echo_owner PUBLIC "${XZ}" PRIVATE "${SPX}/include")
target_compile_definitions(echo_owner PRIVATE XZ_ECHO_TEST malloc=qa_malloc calloc=qa_calloc free=qa_free)
target_compile_options(echo_owner PRIVATE -O2 -fsanitize=undefined -fno-sanitize-recover=all)
add_executable(test_echo test_echo.c)
target_compile_definitions(test_echo PRIVATE XZ_ECHO_TEST)
target_compile_options(test_echo PRIVATE -UNDEBUG -fsanitize=undefined -fno-sanitize-recover=all)
target_link_options(test_echo PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
target_link_libraries(test_echo PRIVATE echo_owner echo_kernel m)
enable_testing()
add_test(NAME echo_resource_and_signal COMMAND test_echo)
