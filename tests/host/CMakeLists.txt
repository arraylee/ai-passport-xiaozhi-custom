cmake_minimum_required(VERSION 3.18)
project(passport_play_checks LANGUAGES C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SERVICES "${REPO}/components/core/services")
option(PASSPORT_UBSAN "Check undefined behavior in native game and persistence tests" ON)
enable_testing()
function(play_test name)
    add_executable(test_${name} test_${name}.c ${ARGN})
    target_include_directories(test_${name} PRIVATE "${SERVICES}/include" "${REPO}/components/core/ports/include")
    target_link_libraries(test_${name} PRIVATE m)
    if(PASSPORT_UBSAN AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(test_${name} PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
        target_link_options(test_${name} PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
    endif()
    # These are assertion-based behavioral checks, including release builds.
    target_compile_options(test_${name} PRIVATE -UNDEBUG)
    add_test(NAME ${name} COMMAND test_${name})
endfunction()
play_test(answer_picker "${SERVICES}/src/answer_picker.c")
play_test(answer_session "${SERVICES}/src/answer_session.c" "${SERVICES}/src/answer_audio_meter.c")
play_test(pet_wish "${SERVICES}/src/pet_wish.c" "${SERVICES}/src/answer_audio_meter.c")
play_test(bobo "${SERVICES}/src/bobo.c" "${SERVICES}/src/bobo_content.c")
play_test(bobo_save "${SERVICES}/src/bobo.c" "${SERVICES}/src/bobo_content.c" "${SERVICES}/src/bobo_save.c")
play_test(clicker_reports "${SERVICES}/src/clicker_reports.c")
play_test(clicker_pointer "${SERVICES}/src/clicker_reports.c")
play_test(clicker_touch "${SERVICES}/src/clicker_reports.c")
play_test(clicker_gate "${SERVICES}/src/clicker_gate.c" "${SERVICES}/src/clicker_reports.c")
play_test(dodge "${SERVICES}/src/dodge.c")
play_test(play_runtime "${SERVICES}/src/play_runtime.c")
